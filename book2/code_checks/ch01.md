```python
import pandas as pd
from statsmodels.stats.proportion import proportion_confint

v = pd.read_csv("validation.csv")   # columns: id, person, tool, stratum, weight

def measure(k, n, name, threshold):
    lo, hi = proportion_confint(k, n, method="wilson")
    verdict = "PASS" if lo >= threshold else "FAIL"
    print(f"{name}: {k/n:.2f} (95% CI {lo:.2f}-{hi:.2f}) -> {verdict} (rule: lower >= {threshold})")

tp = ((v.person == 1) & (v.tool == 1)).sum()
fp = ((v.person == 0) & (v.tool == 1)).sum()
fn = ((v.person == 1) & (v.tool == 0)).sum()
print("Reference positives in sample:", tp + fn)   # if small, stratify
measure(tp, tp + fp, "Precision", 0.70)
measure(tp, tp + fn, "Recall", 0.90)
```
